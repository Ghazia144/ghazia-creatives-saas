async function callAdminApi(action, payload = {}) {
  const { data: { session } } = await supabaseClient.auth.getSession();
  if (!session) return null;

  const res = await fetch(ADMIN_FUNC_URL, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "Authorization": `Bearer ${session.access_token}`,
      "apikey": SUPABASE_PUBLISHABLE_KEY
    },
    body: JSON.stringify({ action, ...payload })
  });
  
  const json = await res.json();
  if (!res.ok) throw new Error(json.error || "API Error");
  return json;
}

function validateEvent(prefix, label) {
  let ok = true;
  const type = $(prefix + "_type").value;
  const date = $(prefix + "_date").value;
  const venueName = $(prefix + "_venue_name").value.trim();
  const venueAddress = $(prefix + "_venue_address").value.trim();

  if (!type) { setErr(prefix + "_type", "Please choose an event type."); ok = false; }
  if (!date) { setErr(prefix + "_date", "Please choose a date."); ok = false; }
  else if (date < todayStr()) { setErr(prefix + "_date", "Date cannot be in the past."); ok = false; }

  if (!venueName) { setErr(prefix + "_venue_name", "Venue name is required."); ok = false; }
  if (!venueAddress) { setErr(prefix + "_venue_address", "Venue address is required."); ok = false; }

  if (maps && !/^https:\/\//i.test(maps)) {

async function submitOrder() {
  if (!validateForm()) {
    const firstErr = document.querySelector(".has-error");
    if (firstErr) firstErr.scrollIntoView({ behavior: "smooth", block: "center" });
    return;
  }

  setSubmitting(true);
  try {
    const creds = JSON.parse(localStorage.getItem(CREDS_KEY) || "null");
    const isUpdate = Boolean(creds && creds.public_code && creds.secret_token);

    const payload = {
      action: isUpdate ? "update" : "create",
      design_id: DESIGN_ID,
      plan: PLAN,
      data: getFormData(),
      customer_whatsapp: normalizePkPhone($("customer_whatsapp").value)
    };

    if (isUpdate) {
      payload.public_code = creds.public_code;
      payload.secret_token = creds.secret_token;
    }

    const res = await fetch(`${SUPABASE_FUNCTIONS_URL}/order`, {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "apikey": SUPABASE_PUBLISHABLE_KEY,
        "Authorization": `Bearer ${SUPABASE_PUBLISHABLE_KEY}`
      },
      body: JSON.stringify(payload)
    });

    const json = await res.json().catch(() => ({}));

    if (!res.ok) {
      throw new Error(json.error || "Could not save your order. Please try again.");
    }

    if (!isUpdate) {
      if (!json.public_code || !json.secret_token) {
        throw new Error("Could not save your order. Please try again.");
      }
      localStorage.setItem(CREDS_KEY, JSON.stringify({
        public_code: json.public_code,
        secret_token: json.secret_token
      }));
    }

    window.location.href = `payment.html?c=${encodeURIComponent(json.public_code || creds.public_code)}`;
  } catch (err) {
    setErr("submit", err.message || "Something went wrong. Please try again.");
    setSubmitting(false);
  }
}

// ---------- Image resize ----------
function resizeImage(file) {
  return new Promise((resolve, reject) => {
    if (!file || !file.type.startsWith("image/")) {
      reject(new Error("Please choose an image file."));
      return;
    }

    const url = URL.createObjectURL(file);
    const img = new Image();
    img.onload = () => {
      URL.revokeObjectURL(url);
      const MAX = 1600;
      let w = img.naturalWidth;
      let h = img.naturalHeight;
      if (w > MAX || h > MAX) {
        const ratio = Math.min(MAX / w, MAX / h);
        w = Math.round(w * ratio);
        h = Math.round(h * ratio);
      }

      const canvas = document.createElement("canvas");
      canvas.width = w;
      canvas.height = h;
      const ctx = canvas.getContext("2d");
      ctx.fillStyle = "#ffffff";
      ctx.fillRect(0, 0, w, h);
      ctx.drawImage(img, 0, 0, w, h);

      canvas.toBlob(
        (blob) => {
          if (!blob) {
            reject(new Error("Could not process image."));
            return;
          }
          resolve(blob);
        },
        "image/jpeg",
        0.8
      );
    };
    img.onerror = () => {
      URL.revokeObjectURL(url);
      reject(new Error("Could not read image."));
    };
    img.src = url;
  });
}

async function api(action, payload) {

        <h1 style="color: #B23A48;">Verification Failed</h1>
        <div class="code-badge">Order: ${code}</div>
        <p>We could not verify your payment: <br><br><strong style="color: var(--ink); background: #fdf2f2; padding: 10px; display: block; border-radius: 4px;">"${data.admin_note}"</strong><br> Please submit it again.</p>
      `;
    }

    html += `
      <div class="btn-group">
        <a href="${payUrl}" class="btn btn-primary">Submit Payment</a>
      </div>
      <a href="https://wa.me/${WHATSAPP_SUPPORT_NUMBER}?text=Hi, I need help with order ${code}" class="help-link" target="_blank">Need help? WhatsApp us</a>
    `;
    container.innerHTML = html;
  }

  function renderNotFound() {
    container.innerHTML = `
      <h1>Order Not Found</h1>
      <p>We couldn't find an order with this code. Please double-check the link.</p>
      <div class="btn-group">
        <a href="https://wa.me/${WHATSAPP_SUPPORT_NUMBER}?text=Hi, my order code is ${code || '(unknown)'}" target="_blank" class="btn btn-primary">WhatsApp Us</a>
      </div>
    `;
  }

  function startPolling() {
    pollInterval = setInterval(() => {
      if (document.visibilityState === 'visible') {
        pollCount++;
        if (pollCount >= MAX_POLLS) {
          clearInterval(pollInterval);
          const p = container.querySelector('.pulse-ring + h1 + .code-badge + p');
          if (p) p.innerHTML += "<br><br><small>Auto-check paused. Please click Refresh.</small>";
        } else {
          fetchStatus(true);
        }
      }
    }, 15000);
  }

  fetchStatus();
